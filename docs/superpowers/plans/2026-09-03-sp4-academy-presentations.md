import fs from 'node:fs'; import path from 'node:path';
const ROOT = path.resolve(new URL('.', import.meta.url).pathname, '..');
const TRACKS = {
  'solar-fundamentals': { order: 1, icon: '☀️', en: 'Solar Fundamentals', he: 'מבוא לסולאר', th: 'พื้นฐานพลังงานแสงอาทิตย์' },

};
const clean = s => s.replace(/<[^>]+>/g, '').replace(/\s+/g, ' ').trim();
function title(html, lang) {
  const a = html.match(new RegExp(`<h1 class="lesson-title" data-${lang}>([^<]*)</h1>`));
  if (a) return clean(a[1]);
  const h1 = html.match(/<h1[^>]*>([\s\S]*?)<\/h1>/); if (!h1) return '';
  const s = h1[1].match(new RegExp(`<span data-${lang}>([\\s\\S]*?)</span>`));
  return s ? clean(s[1]) : '';
}
const lessons = [];
for (const f of fs.readdirSync(path.join(ROOT, 'courses')).filter(f => /^[a-z-]+-\d{2}\.html$/.test(f)).sort()) {
  const m = f.match(/^(.*)-(\d{2})\.html$/); const html = fs.readFileSync(path.join(ROOT, 'courses', f), 'utf8');
  const en = title(html, 'en'), he = title(html, 'he'), th = title(html, 'th');
  if (!en || !he || !th) throw new Error(`${f}: missing title (en=${!!en} he=${!!he} th=${!!th})`);
  lessons.push({ slug: f.replace('.html', ''), track: m[1], num: Number(m[2]), file: `courses/${f}`, title: { en, he, th } });
}

console.log(`wrote assets/lessons.js — ${lessons.length} lessons`);

// academy/tests/i18n-coverage.mjs — every text block in a lesson must have he+th
import { htmlIn, read, fail } from './_util.mjs';
export const BLOCK = /<(p|li|td|th|h2|h3|h4|blockquote|figcaption)\b([^>]*)>([\s\S]*?)<\/\1>/g;
export const TITLE = /<title([^>]*)>([^<]*)<\/title>/;
export function gaps(html) {
  const body = html.replace(/<script[\s\S]*?<\/script>/g, '').replace(/<style[\s\S]*?<\/style>/g, '');
  const out = [];
  for (const m of body.matchAll(BLOCK)) {
    const [whole, tag, attrs, inner] = m;
    if (/\sdata-(en|he|th)\b/.test(attrs) || /\sdata-(en|he|th)\b/.test(inner)) continue; // translated (sibling or span pattern)
    const text = inner.replace(/<[^>]+>/g, '').trim();
    if (!/[A-Za-z]{4,}/.test(text)) continue; // numbers / symbols only
    out.push({ tag, whole, inner, text });
  }
  const t = body.match(TITLE);
  if (t && !/data-he=/.test(t[1])) out.push({ tag: 'title', whole: t[0], inner: t[2], text: t[2] });
  return out;
}

// usage: GEMINI_API_KEY=... node academy/tools/translate-gaps.mjs [--dry] [course-slug ...]
import fs from 'node:fs'; import path from 'node:path'; import crypto from 'node:crypto';

  const j = await res.json(); return JSON.parse(j.candidates[0].content.parts[0].text);
}

    console.log(`translated ${Math.min(i + 15, todo.length)}/${todo.length}`);
  }
}

    html = html.replace(x.whole, repl);
